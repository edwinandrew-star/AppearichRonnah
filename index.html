alter table public.exam_attempts enable row level security; -- ============================================================ -- ANSWERS -- ============================================================ create table if not exists public.exam_answers ( id uuid primary key default gen_random_uuid(), attempt_id uuid not null references public.exam_attempts(id) on delete cascade, question_id uuid not null references public.exam_questions(id) on delete cascade, selected_option_id uuid references public.exam_options(id), answer_text text, is_correct boolean, points_awarded numeric, unique(attempt_id,question_id) ); alter table public.exam_answers enable row level security; -- ============================================================ -- SECURITY EVENTS -- ============================================================ create table if not exists public.exam_security_events ( id uuid primary key default gen_random_uuid(), attempt_id uuid not null references public.exam_attempts(id) on delete cascade, student_id uuid not null references public.profiles(id) on delete cascade, exam_id uuid not null references public.exams(id) on delete cascade, event_type text not null, penalty_minutes integer not null default 5, created_at timestamptz not null default now(), metadata jsonb default '{}'::jsonb ); alter table public.exam_security_events enable row level security; -- ============================================================ -- HELPER: CURRENT USER ROLE -- ============================================================ create or replace function public.current_user_role() returns text language sql stable security definer set search_path = public as $$ select role from public.profiles where id = auth.uid() $$; drop policy if exists "teachers_manage_questions" on public.exam_questions; create policy "teachers_manage_questions" on public.exam_questions for all to authenticated using ( exists ( select 1 from public.exams e where e.id = exam_questions.exam_id and e.teacher_id = auth.uid() ) ) with check ( exists ( select 1 from public.exams e where e.id = exam_questions.exam_id and e.teacher_id = auth.uid() ) ); create policy "teachers_manage_options" on public.exam_options for all to authenticated using ( exists ( select 1 from public.exam_questions q join public.exams e on e.id = q.exam_id where q.id = exam_options.question_id and e.teacher_id = auth.uid() ) ) with check ( exists ( select 1 from public.exam_questions q join public.exams e on e.id = q.exam_id where q.id = exam_options.question_id and e.teacher_id = auth.uid() ) ); -- ============================================================ -- PERMANENT UNLOCK POLICY -- ============================================================ drop policy if exists "students_view_own_unlocks" on public.exam_unlocks; create policy "students_view_own_attempts" on public.exam_attempts for select to authenticated using ( student_id = auth.uid() ); -- ============================================================ -- ANSWERS -- ============================================================ drop policy if exists "students_manage_own_answers" on public.exam_answers; create policy "students_manage_own_answers" on public.exam_answers for all to authenticated using ( exists ( select 1 from public.exam_attempts a where a.id = exam_answers.attempt_id and a.student_id = auth.uid() and a.status = 'IN_PROGRESS' and now() < a.expires_at ) ) with check ( exists ( select 1 from public.exam_attempts a where a.id = exam_answers.attempt_id and a.student_id = auth.uid() and a.status = 'IN_PROGRESS' and now() < a.expires_at ) ); -- ============================================================ -- SECURITY EVENTS -- ============================================================ drop policy if exists "students_view_own_security_events" on public.exam_security_events; create policy "students_view_own_security_events" on public.exam_security_events for select to authenticated using ( student_id = auth.uid() ); -- ============================================================ -- START ATTEMPT -- -- Server-side timing: -- started_at = database NOW() -- expires_at = NOW() + exam duration -- ============================================================ create or replace function public.start_exam_attempt( p_exam_id uuid ) returns uuid language plpgsql security definer set search_path = public as $$ declare v_attempt_id uuid; v_duration integer; begin if public.current_user_role() <> 'student' then raise exception 'Only students can start examinations'; end if; select duration_minutes into v_duration from public.exams where id = p_exam_id and status = 'OPEN'; if v_duration is null then raise exception 'Examination is not open'; end if; if not exists ( select 1 from public.exam_unlocks where exam_id = p_exam_id and student_id = auth.uid() ) then raise exception 'Examination has not been unlocked'; end if; select id into v_attempt_id from public.exam_attempts where exam_id = p_exam_id and student_id = auth.uid(); if v_attempt_id is not null then return v_attempt_id; end if; insert into public.exam_attempts( exam_id, student_id, started_at, expires_at, status ) values( p_exam_id, auth.uid(), now(), now() + make_interval(mins => v_duration), 'IN_PROGRESS' ) returning id into v_attempt_id; return v_attempt_id; end; $$; -- ============================================================ -- LOCK AFTER SECURITY VIOLATION -- -- ALWAYS deduct/track the penalty on the server. -- ============================================================ create or replace function public.lock_exam_for_security_violation( p_attempt_id uuid, p_event_type text ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_attempt public.exam_attempts; v_new_expiry timestamptz; begin select * into v_attempt from public.exam_attempts where id = p_attempt_id and student_id = auth.uid() for update; if v_attempt.id is null then raise exception 'Attempt not found'; end if; if v_attempt.status <> 'IN_PROGRESS' then raise exception 'Attempt is not active'; end if; if now() >= v_attempt.expires_at then raise exception 'Examination has already expired'; end if; insert into public.exam_security_events( attempt_id, student_id, exam_id, event_type, penalty_minutes, metadata ) values( v_attempt.id, v_attempt.student_id, v_attempt.exam_id, p_event_type, 5, jsonb_build_object( 'detected_at',now() ) ); /* The 5-minute deduction is applied when the teacher unlocks, not when the violation occurs. */ update public.exam_attempts set status='LOCKED', violation_count=violation_count+1 where id=v_attempt.id; return jsonb_build_object( 'status','LOCKED', 'expires_at',v_attempt.expires_at ); end; $$; -- ============================================================ -- TEACHER UNLOCK -- -- EXACTLY 5 MINUTES ARE REMOVED. -- SAME ATTEMPT. -- SAME ANSWERS. -- SAME EXAM. -- NO RESET. -- ============================================================ create or replace function public.teacher_unlock_exam_attempt( p_attempt_id uuid ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_attempt public.exam_attempts; v_exam public.exams; v_new_expiry timestamptz; begin if public.current_user_role() <> 'teacher' then raise exception 'Only teachers can unlock examinations'; end if; select * into v_attempt from public.exam_attempts where id=p_attempt_id for update; if v_attempt.id is null then raise exception 'Attempt not found'; end if; select * into v_exam from public.exams where id=v_attempt.exam_id; if v_exam.teacher_id <> auth.uid() then raise exception 'You do not own this examination'; end if; if v_attempt.status <> 'LOCKED' then raise exception 'Attempt is not locked'; end if; v_new_expiry = v_attempt.expires_at - interval '5 minutes'; /* If the deduction would place expiry in the past, the attempt is expired instead of being reopened. */ if v_new_expiry <= now() then update public.exam_attempts set expires_at=now(), status='EXPIRED', penalty_minutes=penalty_minutes+5 where id=v_attempt.id; return jsonb_build_object( 'status','EXPIRED', 'expires_at',now(), 'penalty_minutes',v_attempt.penalty_minutes+5 ); end if; update public.exam_attempts set expires_at=v_new_expiry, status='IN_PROGRESS', penalty_minutes=penalty_minutes+5 where id=v_attempt.id; return jsonb_build_object( 'status','IN_PROGRESS', 'expires_at',v_new_expiry, 'penalty_minutes',v_attempt.penalty_minutes+5 ); end; $$; -- ============================================================ -- SUBMIT EXAM -- -- Correct answers are evaluated server-side. -- ============================================================ create or replace function public.submit_exam_attempt( p_attempt_id uuid ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_attempt public.exam_attempts; v_score numeric := 0; begin select * into v_attempt from public.exam_attempts where id=p_attempt_id and student_id=auth.uid() for update; if v_attempt.id is null then raise exception 'Attempt not found'; end if; if v_attempt.status not in ('IN_PROGRESS','LOCKED') then return jsonb_build_object( 'status',v_attempt.status, 'score',v_attempt.score ); end if; /* Score multiple-choice answers server-side. */ update public.exam_answers a set is_correct = exists ( select 1 from public.exam_options o where o.id=a.selected_option_id and o.question_id=a.question_id and o.is_correct=true ), points_awarded = case when exists ( select 1 from public.exam_options o where o.id=a.selected_option_id and o.question_id=a.question_id and o.is_correct=true ) then ( select q.points from public.exam_questions q where q.id=a.question_id ) else 0 end where a.attempt_id=v_attempt.id; select coalesce(sum(points_awarded),0) into v_score from public.exam_answers where attempt_id=v_attempt.id; update public.exam_attempts set score=v_score, submitted_at=now(), status= case when now() >= expires_at then 'EXPIRED' else 'SUBMITTED' end where id=v_attempt.id; return jsonb_build_object( 'status', case when now() >= v_attempt.expires_at then 'EXPIRED' else 'SUBMITTED' end, 'score',v_score ); end; $$; -- ============================================================ -- IMPORTANT GRANTS -- ============================================================ grant execute on function public.start_exam_attempt(uuid) to authenticated; grant execute on function public.lock_exam_for_security_violation(uuid,text) to authenticated; grant execute on function public.teacher_unlock_exam_attempt(uuid) to authenticated; grant execute on function public.submit_exam_attempt(uuid) to authenticated;
